# NetPulse 本地 GUI 壳 (v1.21.0) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 NetPulse 新增一层零依赖的本地 GUI 壳——双击 exe 弹出 Chromium `--app` 应用窗口，点场景卡诊断、窗口内看进度与报告、一键导出；终端模式经 `--cli` 完整保留。

**Architecture:** 标准库 `http.server`（ThreadingHTTPServer，仅绑 127.0.0.1 + 一次性 token）+ 单页 HTML/CSS/JS（模块常量，~1s 轮询 JSON API）；诊断走既有管线 `run_diagnostics → diagnose → build_report`，经新增 `progress_cb` 参数把模块起止事件写进线程安全快照 `_GuiSession`。新代码不直接读写全局 `LAST_RUN`（收口规则），数据一律取自 `run_diagnostics` 返回的 run dict。

**Tech Stack:** Python 3.13 标准库（http.server / threading / secrets / hmac / ctypes / subprocess / webbrowser）；测试 `py -m pytest`（unittest.mock 模式，参照 `tests/test_v160_scene.py`）。

**视觉基准:** `design_ref/gui/gui_design_v1.html` **方案A 亮色仪表盘**（Simon 已选型）。CSS 变量、卡片/步骤条/亮灯网格的样式与结构从该文件第 1/3/4/5 节提取；本计划给出类名契约与 JS 逻辑全文。

## Global Constraints

- 单文件不变：所有新代码进 `netpulse.py`；页面资产为模块级常量字符串（与 `CSS_CUSTOMER_REPORT` 同惯例）。
- 零新增第三方依赖；只用标准库。
- 新代码不得直接读写全局 `LAST_RUN`（收口规则 2）。允许调用内部会读 `LAST_RUN` 的既有函数（`build_report` / `export_report` / `_site_report_base`），这是沿用管线，不是新增依赖。
- 状态色唯一来源：服务端从 `STATUS_COLORS`/`STATUS_KEY`（netpulse.py:20122-20125）组装进 API 载荷；页面 JS 只消费载荷，不得内嵌第二份色表。超时=中性灰 `#334155`（不是红）。
- 动态数据一律 `textContent` / DOM API 渲染，**禁止 innerHTML 插值动态串**；iframe src 的 URL 由服务端拼接时过 `_html_attr`。
- 终端行为零变化：`progress_cb=None`（默认）时三条既有路径（`--diagnose`/modules/菜单）逐字节不变；全量 pytest 保持绿（约 950+ 用例，`py -m pytest -q`，本机无 `python` 命令，用 `py`）。
- 界面文案大白话（装维可读），按钮字短；界面零外链资源。
- `--json` 永不进 GUI；非 TTY 无参数维持旧行为（进菜单→EOF 退出）。
- 每个任务结束跑全量测试 + commit。commit message 用中文，格式对齐 `git log` 现有风格（`feat: … (v1.21.0)`）。

验证点:
  P1 并行路径: 每模块 start/done 各一条, done 带中文 status
  P2 顺序路径: 事件按 keys 顺序成对出现
  P3 progress_cb=None: 行为与旧版一致 (不抛异常, results 正确)
  P4 回调抛异常不中断诊断 (事件通知永不打断主流程)

import os
import sys
import unittest
from unittest import mock

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, ROOT)
import netpulse as np

_KEYS = ["gateway", "dns", "wifi"]


def _fake_probe(key, cb=None):
    return "完成", {"fake": key}


class TestGuiProgress(unittest.TestCase):

    def _run(self, keys=_KEYS, parallel=True, **kw):
        """跑一次 run_diagnostics, 系统信息/探测全 mock (不碰真实网络)。"""
        def _s(*a, **k):
            return "192.168.1.2"
        with mock.patch.object(np, "_run_module_with_timeout",
                               side_effect=_fake_probe), \
             mock.patch.object(np, "get_local_ip", side_effect=_s), \
             mock.patch.object(np, "get_default_gateway", side_effect=_s), \
             mock.patch.object(np, "get_dns_servers", side_effect=lambda: ["223.5.5.5"]), \
             mock.patch.object(np, "get_public_ip", side_effect=_s), \
             mock.patch.object(np, "get_public_ipv6", side_effect=lambda: None), \
             mock.patch.object(np, "get_ip_geo", side_effect=lambda *a, **k: None):
            return np.run_diagnostics(keys, banner=False, parallel=parallel,
                                      max_workers=2, print_modules=False, **kw)

    def test_parallel_events_paired(self):
        events = []
        run = self._run(parallel=True, progress_cb=events.append)
        starts = sorted(e["key"] for e in events if e["event"] == "start")
        dones = sorted(e["key"] for e in events if e["event"] == "done")
        self.assertEqual(starts, sorted(_KEYS))
        self.assertEqual(dones, sorted(_KEYS))
        for e in events:
            if e["event"] == "done":
                self.assertEqual(e["status"], "完成")
        self.assertEqual(run["status"], {k: "完成" for k in _KEYS})

    def test_sequential_events_in_order(self):
        events = []
        self._run(parallel=False, progress_cb=events.append)
        seq = [(e["event"], e["key"]) for e in events]
        expect = []
        for k in _KEYS:
            expect += [("start", k), ("done", k)]
        self.assertEqual(seq, expect)

    def test_no_callback_unchanged(self):
        run = self._run()
        self.assertEqual(run["status"], {k: "完成" for k in _KEYS})

    def test_callback_exception_ignored(self):
        def _bad(ev):
            raise RuntimeError("gui down")
        run = self._run(progress_cb=_bad)
        self.assertEqual(run["status"], {k: "完成" for k in _KEYS})


if __name__ == "__main__":
    unittest.main()

                "phase": self._phase,
                "profile": self._profile,
                "label": _GUI_LABELS(self._profile) if self._profile else "",
                "total": len(self._keys),
                "done": done,

                "module_status": dict(self._module_status),
                "elapsed_ms": (int((time.time() - self._started_at) * 1000)
                               if self._started_at else 0),
                "error": self._error,
                "has_report": self._report is not None,
                "files": list(self._files),

def _req(port, method, path, token=None, body=None, headers=None):
    conn = http.client.HTTPConnection("127.0.0.1", port, timeout=10)
    h = dict(headers or {})
    if token:

    raw = r.read()

class _GuiHandler(http.server.BaseHTTPRequestHandler):
    """GUI 请求处理器。session/token/shutdown_evt 由 _start_gui_server 注入。"""

    server_version = "NetPulseGUI/" + APP_VERSION
    session = None
    token = None
    shutdown_evt = None

    def log_message(self, fmt, *args):     # GUI 模式终端不刷 HTTP 日志
        pass

    # -- 鉴权与应答工具 --
    def _authorized(self):
        q = urllib_parse.urlparse(self.path)
        tok = (urllib_parse.parse_qs(q.query).get("t") or [None])[0]
        if not tok:
            tok = self.headers.get("X-NetPulse-Token")
        return (isinstance(tok, str) and isinstance(self.token, str)
                and hmac.compare_digest(tok, self.token))

    def _send(self, code, body, ctype="application/json; charset=utf-8"):
        data = body if isinstance(body, bytes) else \
            json.dumps(body, ensure_ascii=False).encode("utf-8")
        self.send_response(code)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(data)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        try:
            self.wfile.write(data)
        except Exception:
            pass

    def _reject(self):
        self._send(403, {"error": "forbidden"})

    def _body(self):
        n = int(self.headers.get("Content-Length") or 0)
        raw = self.rfile.read(n) if n else b""
        if not raw:
            return {}
        try:
            return json.loads(raw.decode("utf-8"))
        except Exception:
            return None

    def _path(self):
        return urllib_parse.urlparse(self.path).path

    # -- 路由 --
    def do_GET(self):
        try:
            if not self._authorized():
                return self._reject()
            p = self._path()
            if p in ("/", "/index.html"):
                return self._send(200, _gui_page().encode("utf-8"),
                                  "text/html; charset=utf-8")
            if p == "/api/scenarios":
                return self._send(200, _gui_meta())
            if p == "/api/progress":
                return self._send(200, self.session.snapshot())
            if p == "/api/report":
                html = self.session.report_html()
                if not html:
                    return self._send(404, {"error": "尚无报告"})
                return self._send(200, html.encode("utf-8"),
                                  "text/html; charset=utf-8")
            return self._send(404, {"error": "not found"})
        except Exception as e:
            try:
                return self._send(500, {"error": str(e)})
            except Exception:
                pass

    def do_POST(self):
        try:
            if not self._authorized():
                return self._reject()
            p, data = self._path(), self._body()
            if data is None:
                return self._send(400, {"error": "bad json"})
            if p == "/api/run":
                profile = str((data or {}).get("profile") or "")
                if self.session.is_running():
                    return self._send(409, {"ok": False, "reason": "running"})
                ok, err = self.session.start(profile)
                return self._send(200 if ok else 400,
                                  {"ok": ok, "err": err})
            if p == "/api/export":
                kind = str((data or {}).get("kind") or "")
                path, err = self.session.export(kind)
                return self._send(200 if path else 400,
                                  {"path": path, "err": err})
            if p == "/api/open-folder":
                try:
                    os.startfile(_desktop_netpulse_dir() or _report_dir())
                    return self._send(200, {"ok": True})
                except Exception as e:
                    return self._send(400, {"ok": False, "err": str(e)})
            if p == "/api/shutdown":
                force = bool((data or {}).get("force"))
                if not self.session.is_running() or force:
                    self.shutdown_evt.set()
                    return self._send(200, {"ok": True})
                return self._send(409, {"ok": False, "reason": "running"})
            # site 端点在 Task 5 追加
            return self._send(404, {"error": "not found"})
        except Exception as e:
            try:
                return self._send(500, {"error": str(e)})
            except Exception:
                pass


def _start_gui_server(session=None, shutdown_evt=None):
    """起本地 GUI 服务: 仅绑 127.0.0.1 随机端口, 返回 (server, port, token)。"""
    import http.server
    import secrets
    session = session if session is not None else _GuiSession()
    shutdown_evt = shutdown_evt if shutdown_evt is not None else threading.Event()
    token = secrets.token_urlsafe(24)
    srv = http.server.ThreadingHTTPServer(("127.0.0.1", 0), _GuiHandler)
    srv.daemon_threads = True
    _GuiHandler.session = session
    _GuiHandler.token = token
    _GuiHandler.shutdown_evt = shutdown_evt
    threading.Thread(target=srv.serve_forever, kwargs={"poll_interval": 0.2},
                     daemon=True, name="np-gui-http").start()
    return srv, srv.server_address[1], token

def _gui_console_process_count():
    """当前控制台挂着的进程数 (双击 exe = 1; 从终端启动 = ≥2)。非 Windows 返回 0。"""
    if sys.platform != "win32":
        return 0
    try:
        import ctypes
        arr = (ctypes.c_ulong * 1)()
        n = ctypes.windll.kernel32.GetConsoleProcessList(arr, 1)
        return int(n or 0)
    except Exception:
        return 0


def _console_is_exclusive():
    """控制台是否被本进程独占 (只有独占才允许隐藏宿主窗口, 不藏用户终端)。"""
    return _gui_console_process_count() == 1


def _console_hide():
    if not _console_is_exclusive():
        return
    try:
        import ctypes
        hwnd = ctypes.windll.kernel32.GetConsoleWindow()
        if hwnd:
            ctypes.windll.user32.ShowWindow(hwnd, 0)   # SW_HIDE
    except Exception:
        pass


def _gui_launch_browser(url):
    """三级降级拉起界面: Chromium --app → 默认浏览器 → "none" (终端打印 URL)。"""
    browser = None
    try:
        browser = _find_pdf_browser()
    except Exception:
        browser = None
    if browser:
        try:
            proc = subprocess.Popen([browser, "--app=" + url,
                                     "--window-size=1280,860"])
            _GUI_BROWSER_PROC.append(proc)      # 保留句柄防 GC
            return "app"
        except OSError:
            pass
    try:
        if webbrowser.open(url):
            return "tab"
    except Exception:
        pass
    return "none"


def _run_gui(install=True, pip_mirror=None, shutdown_evt=None):
    """GUI 主入口 (v1.21.0): 起服务 → 拉起窗口 → 等退出事件。

    install 默认 True: GUI 无人值守, 依赖预检 (scapy/Npcap) 不能弹 input()
    问询 —— 会话卡死在隐藏控制台里没人能答。
    """
    session = _GuiSession()
    shutdown_evt = shutdown_evt if shutdown_evt is not None else threading.Event()
    try:
        srv, port, token = _start_gui_server(session, shutdown_evt)
    except OSError as e:
        print(_c("  ✗ 图形界面服务启动失败 (%s)。" % e, C_RED))
        return
    url = "http://127.0.0.1:%d/?t=%s" % (port, token)
    how = _gui_launch_browser(url)
    if how == "none":
        print(_c("  未检测到可用浏览器。请在任意浏览器打开以下地址使用图形界面:",
                 C_YELLOW))
        print(_c("  " + url, C_WHITE))
        print(_c("  (完成诊断后可关闭本窗口; Ctrl+C 退出)", C_GRAY))
    else:
        _console_hide()      # app/tab 成功 → 双击场景隐藏宿主黑框
    try:
        while not shutdown_evt.is_set():
            shutdown_evt.wait(1.0)
    except KeyboardInterrupt:
        pass
    finally:
        try:
            srv.shutdown()
            srv.server_close()
        except Exception:
            pass


def _GUI_RESET_FOR_TEST():
    """测试隔离: 清 GUI 模块级状态 (不碰业务全局)。"""
    _GUI_SHUTDOWN.clear()
    _GuiHandler.session = None
    _GuiHandler.token = None
    _GuiHandler.shutdown_evt = None
    del _GUI_BROWSER_PROC[:]
